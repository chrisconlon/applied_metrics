Reproduce: the Bayes / rare disease calculation

A test has a 5% false positive rate and a 0% false negative rate. If 1% of the population has the disease, what is the probability a positive test means disease?

posterior <- function(prevalence, fpr = 0.05, fnr = 0.00) {
  p_pos <- (1 - fnr) * prevalence + fpr * (1 - prevalence)
  (1 - fnr) * prevalence / p_pos
}
posterior(0.01)     # 1% prevalence
## [1] 0.1680672
posterior(0.0005)   # 0.05% prevalence: "think horses, not zebras"
## [1] 0.009905894

Reproduce: LLN and CLT histograms

The Bernoulli sample-mean histograms from Lecture 1b:

lln_plot <- function(n, p = 0.4) {
  means <- rbinom(1000, n, p) / n
  hist(means, xlim = c(0, 1), col = "beige", main = paste0("n = ", n),
       xlab = "Sample mean")
  abline(v = p, lwd = 3, col = "red")
}
par(mfrow = c(1, 3))
for (n in c(10, 100, 1000)) lln_plot(n)

clt_plot <- function(n, p = 0.4) {
  z <- (rbinom(1000, n, p) / n - p) * sqrt(n)
  hist(z, xlim = c(-2, 2), col = "beige", freq = FALSE,
       main = paste0("standardized, n = ", n), xlab = "")
  curve(dnorm(x, sd = sqrt(p * (1 - p))), add = TRUE, lwd = 2, col = "red")
}
par(mfrow = c(1, 3))
for (n in c(10, 100, 1000)) clt_plot(n)
