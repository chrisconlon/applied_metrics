Part 1: The two-by-two (Lecture 9a)

Reproduce: a SOX-404-style mandate

Firms with public float of at least $75M (“accelerated filers”) must comply from fiscal 2004; smaller firms are exempt. The outcome is log audit fees. The true effect is \(+0.35\) in 2004, decaying to \(+0.20\) by 2007. Larger firms are on a steeper audit-fee trend: kappa is the extra growth per year per standard deviation of size.

years <- 2000:2007
tau_sox <- c(`2004` = 0.35, `2005` = 0.28, `2006` = 0.22, `2007` = 0.20)
sim_sox <- function(N = 2000, kappa = 0.06, sd_eps = 0.12) {
  lfloat <- rnorm(N, log(75), 1.2)
  size <- (lfloat - mean(lfloat)) / sd(lfloat)
  d <- CJ(firm = 1:N, year = years)
  d[, `:=`(size = size[firm], acc = as.integer(lfloat[firm] >= log(75)))]
  d[, alpha := 11 + 0.5 * size + rnorm(N, 0, 0.4)[firm]]
  gam <- cumsum(c(0, rnorm(length(years) - 1, 0.03, 0.02)))
  d[, post := as.integer(year >= 2004)]
  d[, tau := 0]
  d[acc == 1 & post == 1, tau := tau_sox[as.character(year)]]
  d[, y := alpha + gam[year - 1999] + kappa * size * (year - 2003) + tau + rnorm(.N, 0, sd_eps)]
  d[, D := acc * post]
  d[]
}
sox <- sim_sox()

Reproduce: placebo laws (Bertrand, Duflo and Mullainathan 2004)

A state-by-year panel with AR(1) errors. Half the states “pass a law” in a random year between 6 and 15. The law does nothing. How often does a 5% test reject?

bdm_once <- function(S = 50, T = 20, rho = 0.8) {
  e <- matrix(0, S, T); e[, 1] <- rnorm(S, 0, 1 / sqrt(1 - rho^2))
  for (t in 2:T) e[, t] <- rho * e[, t - 1] + rnorm(S)
  d <- data.table(state = rep(1:S, T), t = rep(1:T, each = S), e = as.vector(e))
  start <- rep(Inf, S); tr <- sample(S, S / 2); start[tr] <- sample(6:15, length(tr), replace = TRUE)
  d[, y := rnorm(S)[state] + 0.1 * t + e]
  d[, D := as.integer(t >= start[state])]
  f <- feols(y ~ D | state + t, d)
  sapply(list(iid = "iid", hetero = "hetero", cluster = ~state),
         function(v) unname(pvalue(summary(f, vcov = v))["D"]) < 0.05)
}
rowMeans(replicate(200, bdm_once(S = 50, rho = 0.8)))
##     iid  hetero cluster 
##   0.370   0.370   0.015

Part 2: Staggered adoption (Lecture 9b)

Reproduce: an algorithm rolled out city by city

300 cities, 20 months, four launch waves (months 6, 9, 12, 15) and a never-treated group, 60 cities each. The effect ramps in over eight months and is larger for earlier waves.

T_ <- 20; waves <- c(6, 9, 12, 15)
sim_rollout <- function(theta = c(`6` = 0.12, `9` = 0.09, `12` = 0.06, `15` = 0.04), ramp = 8, n_per = 60,
                        groups = c(waves, Inf)) {
  G <- rep(groups, each = n_per); N <- length(G)
  d <- CJ(id = 1:N, t = 1:T_)
  d[, g := G[id]]
  d[, alpha := rnorm(N, 0, 0.3)[id]]
  gam <- cumsum(rnorm(T_, 0.01, 0.02))
  d[, e := t - g]
  d[, D := as.integer(t >= g)]
  d[, tau := 0]
  d[D == 1, tau := theta[as.character(g)] * pmin(e + 1, ramp) / ramp]
  d[, y := alpha + gam[t] + tau + rnorm(.N, 0, 0.05)]
  d[, g_sa := ifelse(is.finite(g), g, 10000)]   # sunab's coding for never-treated
  d[]
}
d <- sim_rollout()

Callaway and Sant’Anna by hand: each \(\text{ATT}(g,t)\) is a \(2\times2\) against a clean comparison group, relative to month \(g-1\); then average with cohort-size weights.

cs_att <- function(d, control = "never") {
  wd <- dcast(d, id + g ~ t, value.var = "y"); Y <- as.matrix(wd[, -(1:2)]); G <- wd$g
  out <- list()
  for (g in waves) {
    ig <- which(G == g); b <- g - 1
    for (tt in g:T_) {
      ic <- if (control == "never") which(!is.finite(G)) else which(G > tt & G != g)
      if (length(ic) == 0) next
      out[[length(out) + 1]] <- data.table(g = g, t = tt, n = length(ig),
        att = mean(Y[ig, tt] - Y[ig, b]) - mean(Y[ic, tt] - Y[ic, b]))
    }
  }
  a <- rbindlist(out)
  a[, weighted.mean(att, n)]
}
c(cs_never = cs_att(d, "never"), cs_notyet = cs_att(d, "notyet"))
