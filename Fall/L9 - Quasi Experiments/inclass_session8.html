Part 1: Panel data and fixed effects

Reproduce: the management-practices panel

Firm quality \(\alpha_i\) raises productivity and makes firms adopt better practices. The true effect of practices on log productivity is \(\beta = 0.10\).

beta_true <- 0.10
sim_panel <- function(N = 200, T = 8, rho = 0.7, sd_alpha = 0.5, sd_eps = 0.3, me_sd = 0) {
  alpha <- rnorm(N, 0, sd_alpha)
  gamma <- rnorm(T, 0, 0.1)
  d <- CJ(firm = 1:N, year = 1:T)
  d[, alpha := alpha[firm]]
  d[, x := rho * alpha + sqrt(1 - rho^2) * (rnorm(N)[firm] * 0.6) + rnorm(.N, 0, 0.8)]
  d[, y := 1 + beta_true * x + alpha + gamma[year] + rnorm(.N, 0, sd_eps)]
  if (me_sd > 0) d[, x := x + rnorm(.N, 0, me_sd)]
  d[]
}
d <- sim_panel()

Reproduce: random effects by hand

Random effects is GLS with equicorrelated errors, which is OLS after quasi-demeaning with weight \(\theta = 1 - \sqrt{\sigma^2_\varepsilon / (\sigma^2_\varepsilon + T\sigma^2_\alpha)}\). Estimate the two variances from the within and between regressions (Swamy–Arora), then transform.

re_fit <- function(d) {
  within <- feols(y ~ x | firm, d)
  s2_eps <- sum(resid(within)^2) / (nrow(d) - uniqueN(d$firm) - 1)
  b <- d[, .(ybar = mean(y), xbar = mean(x), T_i = .N), by = firm]
  between <- lm(ybar ~ xbar, b)
  s2_alpha <- max(0, sum(resid(between)^2) / between$df.residual - s2_eps / mean(b$T_i))
  d <- merge(d, b, by = "firm")
  d[, theta := 1 - sqrt(s2_eps / (s2_eps + T_i * s2_alpha))]
  fit <- lm(I(y - theta * ybar) ~ I(x - theta * xbar), d)
  list(beta = unname(coef(fit)[2]), se = unname(sqrt(diag(vcov(fit)))[2]),
       theta = mean(d$theta), s2_eps = s2_eps, s2_alpha = s2_alpha)
}
re <- re_fit(d)

Part 2: Event studies

Reproduce: the single-timing event study

400 firms, 24 quarters. Half adopt a pricing tool in quarter 13; the effect ramps from 0.0125 to 0.05 over four quarters. Never-treated firms get the reference value of event time so they stay in the regression and identify the period effects.

tau_true <- function(k) ifelse(k >= 0, 0.05 * pmin(1, (k + 1) / 4), 0)
sim_es <- function(N = 400, T = 24, E0 = 13, trend = 0, anticipate = 0, sd_eps = 0.08) {
  d <- CJ(firm = 1:N, t = 1:T)
  d[, treated := as.integer(firm <= N / 2)]
  d[, alpha := rnorm(N, 0, 0.3)[firm]]
  gam <- cumsum(rnorm(T, 0.005, 0.02))
  d[, k := ifelse(treated == 1, t - E0, -1L)]          # controls carry the reference value
  d[, tau := ifelse(treated == 1, tau_true(t - E0), 0)]
  if (anticipate > 0) d[treated == 1 & (t - E0) < 0 & (t - E0) >= -anticipate, tau := 0.0125]
  d[, y := alpha + gam[t] + tau + trend * treated * (t - E0) + rnorm(.N, 0, sd_eps)]
  d[]
}
es <- sim_es()
fit_es <- feols(y ~ i(k, ref = -1) | firm + t, es, cluster = ~firm)
iplot(fit_es, ref.line = -0.5, main = "Event study: log gross margin around adoption")
lines(-12:11, tau_true(-12:11), col = "red", lwd = 2)

Part 3: The returns event study

Reproduce: earnings announcements and CARs

200 firms, one announcement each, 261 trading days per firm. The market model is estimated on days \(-250\) to \(-11\); abnormal returns are computed on \(-10\) to \(+10\). A one-SD earnings surprise moves the stock 2% on day 0 and drifts another 1% over the next ten days.

N_c <- 200; days <- -250:10
S <- rnorm(N_c)
car <- CJ(firm = 1:N_c, day = days)
car[, `:=`(S = S[firm], beta = rnorm(N_c, 1, 0.3)[firm], a = rnorm(N_c, 0, 2e-4)[firm],
           Rm = rnorm(.N, 4e-4, 0.010))]        # each firm on its own calendar dates
car[, ar_true := fifelse(day == 0, 0.02 * S, fifelse(day >= 1 & day <= 10, 0.001 * S, 0))]
car[, R := a + beta * Rm + ar_true + rnorm(.N, 0, 0.02)]

est <- car[day <= -11, { f <- lm(R ~ Rm); .(a_hat = coef(f)[1], b_hat = coef(f)[2], s_hat = summary(f)$sigma) }, by = firm]
car <- merge(car, est, by = "firm")
car[, AR := R - a_hat - b_hat * Rm]
ev <- car[day >= -10 & day <= 10][, CAR := cumsum(AR), by = firm]
