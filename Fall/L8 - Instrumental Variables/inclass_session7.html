Part 1: LATE in an encouragement design

A retailer emails a random half of its customers to nudge them onto its app. Twenty percent would install it anyway (always-takers), forty percent install it only if emailed (compliers), forty percent never do. The app’s effect on spending differs by type.

sim_late <- function(N = 10000, tau = c(always = 2.0, complier = 0.8, never = 0.5)) {
  type <- sample(names(tau), N, replace = TRUE, prob = c(0.2, 0.4, 0.4))
  Z <- rbinom(N, 1, 0.5)
  D <- as.integer(type == "always" | (type == "complier" & Z == 1))
  base <- c(always = 3, complier = 1, never = 0.5)[type]
  tenure <- c(always = 4, complier = 2, never = 1)[type] + rnorm(N)
  data.frame(y = base + tau[type] * D + rnorm(N), D, Z, tenure, type, tau = tau[type])
}

Part 2: A judge design with refund agents

Requests are routed at random to 50 agents who differ in generosity. The claim’s merit \(q\) is seen by the agent, not by us; refunds build the most goodwill for weak claims.

sim_refund <- function(N = 20000, J = 50, typed = FALSE) {
  q <- rnorm(N); agent <- sample(1:J, N, replace = TRUE)
  ctype <- sample(c("shipping", "damage"), N, replace = TRUE)
  gen <- rnorm(J, -0.5, 0.5)
  if (typed) { gen2 <- -0.5 - (gen + 0.5)                # agents lenient on shipping are strict on damage
    cutoff <- ifelse(ctype == "shipping", gen[agent], gen2[agent]) } else cutoff <- gen[agent]
  R <- as.integer(q + cutoff + rnorm(N, 0, 0.5) > 0)
  tau <- 20 - 15 * q
  d <- data.frame(spend = 100 + tau * R + 30 * q + rnorm(N, 0, 20), R, agent, q, tau, ctype)
  tot <- tapply(R, agent, sum); n_j <- tapply(R, agent, length)
  d$len <- (tot[agent] - R) / (n_j[agent] - 1)           # leave-one-out leniency
  d
}
dr <- sim_refund()
etable(feols(spend ~ R, dr, cluster = ~agent), feols(R ~ len, dr, cluster = ~agent),
       feols(spend ~ 1 | R ~ len, dr, cluster = ~agent), headers = c("OLS", "first stage", "2SLS"))

Part 3: Shift-share with Rotemberg weights

Markets differ in which sector their banks lend to; national sector credit shocks move local credit growth; the question is the effect of local credit on firm entry (truth 0.4).

sim_ss <- function(M = 1000, K = 6) {
  dom <- sample(1:K, M, replace = TRUE); sh <- matrix(0.4 / (K - 1), M, K); sh[cbind(1:M, dom)] <- 0.6
  g <- rnorm(K); eta <- rnorm(M); B <- as.vector(sh %*% g)
  x <- B + 0.5 * eta + rnorm(M, 0, 0.3); y <- 0.4 * x + eta + rnorm(M, 0, 0.3)
  list(d = data.frame(y, x, B), sh = sh, g = g)
}
s <- sim_ss(); d <- s$d
etable(feols(y ~ x, d), feols(y ~ 1 | x ~ B, d, vcov = "hetero"), headers = c("OLS", "2SLS"))

# Rotemberg weights: alpha_k proportional to g_k * sum_m s_mk * (x_m - mean x)
xt <- d$x - mean(d$x)
alpha <- sapply(1:6, function(k) s$g[k] * sum(s$sh[, k] * xt)); alpha <- alpha / sum(alpha)
beta_k <- sapply(1:6, function(k) coef(feols(y ~ 1 | x ~ z, data.frame(d, z = s$sh[, k] * s$g[k])))[2])
round(cbind(shock = s$g, weight = alpha, beta_k = beta_k), 2); sum(alpha * beta_k)
